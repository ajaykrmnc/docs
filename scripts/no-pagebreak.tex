\usepackage{etoolbox}
\usepackage{needspace}
\usepackage{float}

% Keep ASCII diagrams/code blocks together when they fit on one page.
% If there is not enough room left, start the block on the next page.
\BeforeBeginEnvironment{verbatim}{\Needspace{18\baselineskip}\begin{samepage}}
\AfterEndEnvironment{verbatim}{\end{samepage}}
\BeforeBeginEnvironment{Highlighting}{\Needspace{18\baselineskip}\begin{samepage}}
\AfterEndEnvironment{Highlighting}{\end{samepage}}

% Pandoc emits some markdown tables as longtable and some as table/tabular.
% Prefer moving them intact to the next page over splitting them mid-table.
\BeforeBeginEnvironment{longtable}{\Needspace{12\baselineskip}}
\BeforeBeginEnvironment{tabular}{\Needspace{10\baselineskip}\begin{samepage}}
\AfterEndEnvironment{tabular}{\end{samepage}}

% Avoid floating tables/figures drifting far away from surrounding text.
\floatplacement{table}{H}
\floatplacement{figure}{H}
